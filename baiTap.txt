1.  Tìm phần tử có số lần xuất hiện ít nhất trong mảng, nếu có nhiều phần tử có cùng số lần xuất hiện ít nhất, chỉ cần trả về phần tử đầu tiên xuất hiện trong các phần tử đó

Ví dụ: Nếu mảng dữ liệu nhận được là [1, 2, 3, 2, 1], chương trình sẽ tìm ra phần tử xuất hiện ít nhất là 3 (xuất hiện 1 lần), nên mảng gửi lên server là [3, 1].

                }

                    (byte) result,
                    (byte) minCount
            };

            service.submitData(studentCode, qCode, resultData);

        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}

c. Tính giá trị trung bình, độ lệch chuẩn trên toàn bộ dãy và phân vị 95 tại vị trí ceil(n * 0.95) - 1 sau khi sắp xếp tăng dần.

import RMI.DataService;
import java.math.BigDecimal;
import java.math.RoundingMode;
import java.rmi.registry.LocateRegistry;
import java.rmi.registry.Registry;
import java.util.Arrays;

public class tzqzJy7j {

    public static void main(String[] args) {
        String studentCode = "B23DCCE005";
        String qCode = "tzqzJy7j";

        try {
            Registry registry = LocateRegistry.getRegistry(
                    "36.50.135.242", 1099
            );

            DataService service = (DataService)
                    registry.lookup("RMIDataService");

            Object data = service.requestData(studentCode, qCode);

            if (data == null) {
                System.out.println("Server tra ve NULL!");
                return;
            }

            String csv = (String) data;
            System.out.println("Du lieu: " + csv);

            String[] parts = csv.split(",");
            int n = parts.length;

            double[] arr = new double[n];
            BigDecimal sum = BigDecimal.ZERO;

            for (int i = 0; i < n; i++) {
                BigDecimal value = new BigDecimal(parts[i].trim());
                arr[i] = value.doubleValue();
                sum = sum.add(value);
            }

            // Tinh trung binh chinh xac
            BigDecimal avgBD = sum.divide(
                    BigDecimal.valueOf(n),
                    15,
                    RoundingMode.HALF_UP
            );

            double average = avgBD.doubleValue();

            // Tinh do lech chuan tong the
            double variance = 0;
            for (double x : arr) {
                variance += Math.pow(x - average, 2);
            }
            double stddev = Math.sqrt(variance / n);

            // Tinh phan vi 95
            Arrays.sort(arr);
            int index = (int) Math.ceil(n * 0.95) - 1;
            double p95 = arr[index];

            // Lam tron HALF_UP
            String avgStr = avgBD.setScale(
                    2, RoundingMode.HALF_UP
            ).toPlainString();

            String stdStr = BigDecimal.valueOf(stddev)
                    .setScale(2, RoundingMode.HALF_UP)
                    .toPlainString();

            String p95Str = BigDecimal.valueOf(p95)
                    .setScale(2, RoundingMode.HALF_UP)
                    .toPlainString();

            String result = "average=" + avgStr
                    + ";stddev=" + stdStr
                    + ";p95=" + p95Str;

            System.out.println("Ket qua: " + result);

            service.submitData(studentCode, qCode, result);

            System.out.println("Da gui ket qua len server!");

        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}

        this.breached = breached;
    }

    }

    }

import RMI.ObjectService;
import RMI.TicketSla;

import java.rmi.Naming;

public class q7cAs0tuJ {
    public static void main(String[] args) {
        String studentCode = "B23DCCE005";
        String qCode = "7cAs0tuJ";

        try {
            ObjectService service =
                    (ObjectService) Naming.lookup(
                            "rmi://36.50.135.242/RMIObjectService"
                    );

            TicketSla ticket =
                    (TicketSla) service.requestObject(studentCode, qCode);

            String priority = ticket.getPriority();
            int hours = ticket.getOpenedHoursAgo();

            boolean breached = false;

            switch (priority) {
                case "CRITICAL":
                    breached = hours > 2;
                    break;
                case "HIGH":
                    breached = hours > 8;
                    break;
                case "MEDIUM":
                    breached = hours > 24;
                    break;
                case "LOW":
                    breached = hours > 72;
                    break;
            }

            ticket.setBreached(breached);

            if (!breached) {
                ticket.setAction("MONITOR");
            } else if (priority.equals("CRITICAL") || hours > 96) {
                ticket.setAction("ESCALATE_L2");
            } else {
                ticket.setAction("ESCALATE_L1");
            }

            service.submitObject(studentCode, qCode, ticket);

        } catch (Exception e) {
            e.printStackTrace();
        }
    }

import java.io.IOException;
import java.io.InputStream;
import java.io.OutputStream;
import java.net.HttpURLConnection;

import javax.xml.parsers.DocumentBuilderFactory;
import org.w3c.dom.Document;
import org.w3c.dom.NodeList;

public class RriwlduN {

    // ============================================================
    // 1. CONFIGURATION: Cau hinh Web Service
    // ============================================================
    static final String ENDPOINT =
            "http://36.50.135.242:2221/DataService";

    static final String NAMESPACE =
            "http://soap.server.network.judge.dblab.ptit/";

    static final String STUDENT_CODE = "B23DCCE005";
    static final String QUESTION_CODE = "RriwlduN";

    public static void main(String[] args) throws Exception {

        // ========================================================
        // 2. INPUT: Lay du lieu tu server
        // ========================================================
        List<Integer> numbers = getData(STUDENT_CODE, QUESTION_CODE);

        System.out.println("Input data: " + numbers);

        // ========================================================
        // 3. ALGORITHM: Ghep cac so de tao so lon nhat
        // Thay phan nay bang thuat toan cua cau hoi khac.
        // ========================================================
        List<String> parts = new ArrayList<>();

        for (Integer number : numbers) {
            parts.add(String.valueOf(number));
        }

        parts.sort((a, b) -> (b + a).compareTo(a + b));

        StringBuilder answerBuilder = new StringBuilder();

        for (String part : parts) {
            answerBuilder.append(part);
        }

        String answer = answerBuilder.toString();

        System.out.println("Result: " + answer);

        // ========================================================
        // 4. OUTPUT: Gui ket qua ve server
        // ========================================================
        submitDataString(STUDENT_CODE, QUESTION_CODE, answer);

        System.out.println("Submission completed.");
    }

    // ============================================================
    // SOAP HELPER: Gui SOAP request va nhan XML response
    // Ham nay co the tai su dung cho cac bai SOAP cung kieu.
    // ============================================================
    static Document callSoap(String operation, String parametersXml)
            throws Exception {

        String requestXml =
                "<?xml version=\"1.0\" encoding=\"UTF-8\"?>"
                + "<soapenv:Envelope"
                + " xmlns:soapenv=\"http://schemas.xmlsoap.org/soap/envelope/\""
                + " xmlns:tns=\"" + NAMESPACE + "\">"
                + "<soapenv:Header/>"
                + "<soapenv:Body>"
                + "<tns:" + operation + ">"
                + parametersXml
                + "</tns:" + operation + ">"
                + "</soapenv:Body>"
                + "</soapenv:Envelope>";

        HttpURLConnection connection =
                (HttpURLConnection) new URL(ENDPOINT).openConnection();

        connection.setRequestMethod("POST");
        connection.setConnectTimeout(15000);
        connection.setReadTimeout(15000);
        connection.setDoOutput(true);

        connection.setRequestProperty(
                "Content-Type", "text/xml; charset=utf-8");

        connection.setRequestProperty("SOAPAction", "\"\"");

        try {
            // Gui request den Web Service.
            try (OutputStream output = connection.getOutputStream()) {
                output.write(requestXml.getBytes(StandardCharsets.UTF_8));
            }

            int statusCode = connection.getResponseCode();

            InputStream responseStream =
                    (statusCode >= 200 && statusCode < 300)
                    ? connection.getInputStream()
                    : connection.getErrorStream();

            if (responseStream == null) {
                throw new IOException(
                        "No response body. HTTP status: " + statusCode);
            }

            // Parse XML response va vo hieu hoa external entities.
            DocumentBuilderFactory factory =
                    DocumentBuilderFactory.newInstance();

            factory.setNamespaceAware(true);

            factory.setFeature(
                    "http://apache.org/xml/features/disallow-doctype-decl",
                    true);

            factory.setFeature(
                    "http://xml.org/sax/features/external-general-entities",
                    false);

            factory.setFeature(
                    "http://xml.org/sax/features/external-parameter-entities",
                    false);

            Document response;

            try (InputStream input = responseStream) {
                response = factory.newDocumentBuilder().parse(input);
            }

            // Kiem tra loi SOAP tu server.
            NodeList faults =
                    response.getElementsByTagNameNS("*", "Fault");

            if (faults.getLength() > 0) {
                throw new IOException(
                        "SOAP Fault: " + faults.item(0).getTextContent());
            }

            if (statusCode < 200 || statusCode >= 300) {
                throw new IOException("HTTP status: " + statusCode);
            }

            return response;

        } finally {
            connection.disconnect();
        }
    }

    // ============================================================
    // INPUT: getData(studentCode, qCode) -> List<Integer>
    // Doi ham nay neu bai khac co phuong thuc input khac.
    // ============================================================
    static List<Integer> getData(
            String studentCode, String questionCode) throws Exception {

        String parametersXml =
                "<studentCode>" + escapeXml(studentCode)
                + "</studentCode>"
                + "<qCode>" + escapeXml(questionCode)
                + "</qCode>";

        Document response = callSoap("getData", parametersXml);

        NodeList returnNodes =
                response.getElementsByTagNameNS("*", "return");

        List<Integer> numbers = new ArrayList<>();

        for (int i = 0; i < returnNodes.getLength(); i++) {
            String value =
                    returnNodes.item(i).getTextContent().trim();

            if (!value.isEmpty()) {
                numbers.add(Integer.parseInt(value));
            }
        }

        return numbers;
    }

    // ============================================================
    // OUTPUT: submitDataString(studentCode, qCode, data)
    // Doi ham nay neu bai khac yeu cau kieu output khac.
    // ============================================================
    static void submitDataString(
            String studentCode,
            String questionCode,
            String data) throws Exception {

        String parametersXml =
                "<studentCode>" + escapeXml(studentCode)
                + "</studentCode>"
                + "<qCode>" + escapeXml(questionCode)
                + "</qCode>"
                + "<data>" + escapeXml(data)
                + "</data>";

        callSoap("submitDataString", parametersXml);
    }

    // ============================================================
    // UTILITY: Escape ky tu dac biet trong XML
    // ============================================================
    static String escapeXml(String value) {
        return value.replace("&", "&amp;")
                    .replace("<", "&lt;")
                    .replace(">", "&gt;")
                    .replace("\"", "&quot;")
                    .replace("'", "&apos;");
    }
}

import java.io.IOException;

public class WlJ4tCwZ {

    // ============================================================
    // SOAP HELPER: Gui SOAP request va nhan XML response
    // Dung lai ham nay cho cac bai SOAP cung kieu.
    // ============================================================
    static Document callSoap(String operation, String parametersXml)
            throws Exception {

        String requestXml =
                "<?xml version=\"1.0\" encoding=\"UTF-8\"?>"
                + "<soapenv:Envelope"
                + " xmlns:soapenv=\"http://schemas.xmlsoap.org/soap/envelope/\""
                + " xmlns:tns=\"" + NAMESPACE + "\">"
                + "<soapenv:Header/>"
                + "<soapenv:Body>"
                + "<tns:" + operation + ">"
                + parametersXml
                + "</tns:" + operation + ">"
                + "</soapenv:Body>"
                + "</soapenv:Envelope>";

        HttpURLConnection connection =
                (HttpURLConnection) new URL(ENDPOINT).openConnection();

        connection.setRequestMethod("POST");
        connection.setConnectTimeout(15000);
        connection.setReadTimeout(15000);
        connection.setDoOutput(true);

        connection.setRequestProperty(
                "Content-Type", "text/xml; charset=utf-8");

        connection.setRequestProperty("SOAPAction", "\"\"");

        try {
            // Gui request den Web Service.
            try (OutputStream output = connection.getOutputStream()) {
                output.write(requestXml.getBytes(StandardCharsets.UTF_8));
            }

            int statusCode = connection.getResponseCode();

            InputStream responseStream =
                    (statusCode >= 200 && statusCode < 300)
                    ? connection.getInputStream()
                    : connection.getErrorStream();

            if (responseStream == null) {
                throw new IOException(
                        "No response body. HTTP status: " + statusCode);
            }

            // Parse XML response va vo hieu hoa external entities.
            DocumentBuilderFactory factory =
                    DocumentBuilderFactory.newInstance();

            factory.setNamespaceAware(true);

            factory.setFeature(
                    "http://apache.org/xml/features/disallow-doctype-decl",
                    true);

            factory.setFeature(
                    "http://xml.org/sax/features/external-general-entities",
                    false);

            factory.setFeature(
                    "http://xml.org/sax/features/external-parameter-entities",
                    false);

            Document response;

            try (InputStream input = responseStream) {
                response = factory.newDocumentBuilder().parse(input);
            }

            // Kiem tra SOAP Fault.
            NodeList faults =
                    response.getElementsByTagNameNS("*", "Fault");

            if (faults.getLength() > 0) {
                throw new IOException(
                        "SOAP Fault: " + faults.item(0).getTextContent());
            }

            if (statusCode < 200 || statusCode >= 300) {
                throw new IOException("HTTP status: " + statusCode);
            }

            return response;

        } finally {
            connection.disconnect();
        }
    }

    // ============================================================
    // INPUT: getData(studentCode, qCode) -> List<Integer>
    // ============================================================
    static List<Integer> getData(
            String studentCode, String questionCode) throws Exception {

        String parametersXml =
                "<studentCode>" + escapeXml(studentCode)
                + "</studentCode>"
                + "<qCode>" + escapeXml(questionCode)
                + "</qCode>";

        Document response = callSoap("getData", parametersXml);

        NodeList returnNodes =
                response.getElementsByTagNameNS("*", "return");

        List<Integer> numbers = new ArrayList<>();

        for (int i = 0; i < returnNodes.getLength(); i++) {
            String value =
                    returnNodes.item(i).getTextContent().trim();

            if (!value.isEmpty()) {
                numbers.add(Integer.parseInt(value));
            }
        }

        return numbers;
    }

    // ============================================================
    // OUTPUT: submitDataInt(studentCode, qCode, data)
    // ============================================================
    static void submitDataInt(
            String studentCode,
            String questionCode,
            int data) throws Exception {

        String parametersXml =
                "<studentCode>" + escapeXml(studentCode)
                + "</studentCode>"
                + "<qCode>" + escapeXml(questionCode)
                + "</qCode>"
                + "<data>" + data + "</data>";

        callSoap("submitDataInt", parametersXml);
    }

    // ============================================================
    // UTILITY: Escape cac ky tu dac biet trong XML
    // ============================================================
    static String escapeXml(String value) {
        return value.replace("&", "&amp;")
                    .replace("<", "&lt;")
                    .replace(">", "&gt;")
                    .replace("\"", "&quot;")
                    .replace("'", "&apos;");
    }

    // ============================================================
    // XML HELPER: Lay text cua mot phan tu con truc tiep
    // So sanh local name de ho tro ca element co namespace.
    // ============================================================
    static String getChildText(Element parent, String wantedName) {
        NodeList children = parent.getChildNodes();

        for (int i = 0; i < children.getLength(); i++) {
            Node node = children.item(i);

            if (node instanceof Element) {
                Element element = (Element) node;
                String localName = element.getLocalName();

                if (localName == null) {
                    localName = element.getNodeName();
                    int colon = localName.indexOf(':');

                    if (colon >= 0) {
                        localName = localName.substring(colon + 1);
                    }
                }

                if (wantedName.equals(localName)) {
                    return element.getTextContent().strip();
                }
            }
        }

        return null;
    }

                + "</qCode>";

        Document response =
                callSoap("requestListEmployeeY", parametersXml);

        List<EmployeeY> employees = new ArrayList<>();
        NodeList elements = response.getElementsByTagName("*");

        for (int i = 0; i < elements.getLength(); i++) {
            Node node = elements.item(i);

            if (!(node instanceof Element)) {
                continue;
            }

            Element element = (Element) node;

            String name = getChildText(element, "name");
            String startDate = getChildText(element, "startDate");

            // Chi nhan phan tu dai dien cho EmployeeY.
            if (name != null && startDate != null) {
                employees.add(new EmployeeY(name, startDate));
            }
        }

        return employees;

    static String getChildText(Element parent, String wantedName) {
        NodeList children = parent.getChildNodes();

        for (int i = 0; i < children.getLength(); i++) {
            Node node = children.item(i);

            if (!(node instanceof Element)) {
                continue;
            }

            Element element = (Element) node;
            String localName = element.getLocalName();

            if (localName == null) {
                localName = element.getNodeName();
                int colon = localName.indexOf(':');

                if (colon >= 0) {
                    localName = localName.substring(colon + 1);
                }
            }

            if (wantedName.equals(localName)) {
                return element.getTextContent().trim();
            }
        }

        return null;
    }

    static List<Customer> requestListCustomer(
            String studentCode, String qCode) throws Exception {

        String parametersXml =
                "<studentCode>" + escapeXml(studentCode)
                + "</studentCode>"
                + "<qCode>" + escapeXml(qCode) + "</qCode>";

        Document response =
                callSoap("requestListCustomer", parametersXml);

        NodeList elements =
                response.getElementsByTagNameNS("*", "return");

        List<Customer> customers = new ArrayList<>();

        for (int i = 0; i < elements.getLength(); i++) {
            Node node = elements.item(i);

            if (!(node instanceof Element)) {
                continue;
            }

            Element element = (Element) node;

            String customerId = getChildText(element, "customerId");
            String location = getChildText(element, "location");
            String purchaseCountText =
                    getChildText(element, "purchaseCount");
            String totalSpentText =
                    getChildText(element, "totalSpent");

            if (customerId != null && location != null
                    && purchaseCountText != null
                    && totalSpentText != null) {

                customers.add(new Customer(
                        customerId,
                        location,
                        Integer.parseInt(purchaseCountText),
                        Float.parseFloat(totalSpentText)
                ));
            }
        }

        return customers;
    }

        for (Customer customer : customers) {
            parametersXml.append("<data>")
                    .append("<customerId>")

                    .append("<location>")
                    .append(escapeXml(customer.location))
                    .append("</location>")
                    .append("<purchaseCount>")
                    .append(customer.purchaseCount)
                    .append("</purchaseCount>")
                    .append("<totalSpent>")
                    .append(Float.toString(customer.totalSpent))
                    .append("</totalSpent>")
                    .append("</data>");
        }

        callSoap("submitListCustomer", parametersXml.toString());
    }

    static String getChildText(
            Element parent, String name) {

        NodeList children = parent.getChildNodes();

        for (int i = 0; i < children.getLength(); i++) {

            Node node = children.item(i);

            if (node instanceof Element) {
                Element element = (Element) node;

                String localName = element.getLocalName();

                if (localName == null) {
                    localName = element.getNodeName();

                    int colon = localName.indexOf(':');
                    if (colon >= 0) {
                        localName = localName.substring(colon + 1);
                    }
                }

                if (name.equals(localName)) {
                    return element.getTextContent().trim();
                }
            }
        }

        return null;
    }

            for (Customer c : customers) {
                if (c.purchaseCount >= 6
                        && c.totalSpent.compareTo(
                                new BigDecimal("4000")) >= 0) {
                    result.add(c);
                }
            }

            // 3. Sap xep totalSpent giam dan
            // Neu bang nhau: customerId tang dan
            result.sort((a, b) -> {
                int cmp = b.totalSpent.compareTo(a.totalSpent);

                if (cmp != 0) {
                    return cmp;
                }

                return a.customerId.compareTo(b.customerId);
            });

            System.out.println("\nOutput:");
            for (Customer c : result) {
                System.out.println(c);
            }

            // 4. Gui danh sach ve server
            String status = submitListCustomer(
                    STUDENT_CODE, QUESTION_CODE, result);

            System.out.println("\nStatus: " + status);
            System.out.println("Submission completed.");

        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    // =============================================
    // DATA MODEL: Customer
    // =============================================
    static class Customer {
        String customerId;
        int purchaseCount;
        BigDecimal totalSpent;

        Customer(String customerId,
                 int purchaseCount,
                 BigDecimal totalSpent) {

            this.customerId = customerId;
            this.purchaseCount = purchaseCount;
            this.totalSpent = totalSpent;
        }

        @Override
        public String toString() {
            return "Customer{customerId=" + customerId
                    + ", purchaseCount=" + purchaseCount
                    + ", totalSpent="
                    + totalSpent.toPlainString() + "}";
        }
    }

    // =============================================
    // SOAP HELPER
    // =============================================
    static Document callSoap(
            String operation, String parametersXml)
            throws Exception {

        String requestXml =
                "<?xml version=\"1.0\" encoding=\"UTF-8\"?>"
                + "<soapenv:Envelope"
                + " xmlns:soapenv=\"http://schemas.xmlsoap.org/soap/envelope/\""
                + " xmlns:tns=\"" + NAMESPACE + "\">"
                + "<soapenv:Header/>"
                + "<soapenv:Body>"
                + "<tns:" + operation + ">"
                + parametersXml
                + "</tns:" + operation + ">"
                + "</soapenv:Body>"
                + "</soapenv:Envelope>";

        HttpURLConnection conn = (HttpURLConnection)
                new URL(ENDPOINT).openConnection();

        try {
            conn.setRequestMethod("POST");
            conn.setConnectTimeout(15000);
            conn.setReadTimeout(15000);
            conn.setDoOutput(true);

            conn.setRequestProperty(
                    "Content-Type", "text/xml; charset=utf-8");

            conn.setRequestProperty("SOAPAction", "\"\"");

            try (OutputStream os = conn.getOutputStream()) {
                os.write(requestXml.getBytes(
                        StandardCharsets.UTF_8));
            }

            int statusCode = conn.getResponseCode();

            InputStream stream =
                    statusCode >= 200 && statusCode < 300
                    ? conn.getInputStream()
                    : conn.getErrorStream();

            if (stream == null) {
                throw new IOException(
                        "HTTP status: " + statusCode);
            }

            DocumentBuilderFactory factory =
                    DocumentBuilderFactory.newInstance();

            factory.setNamespaceAware(true);

            factory.setFeature(
                    "http://apache.org/xml/features/disallow-doctype-decl",
                    true);

            factory.setFeature(
                    "http://xml.org/sax/features/external-general-entities",
                    false);

            factory.setFeature(
                    "http://xml.org/sax/features/external-parameter-entities",
                    false);

            Document response;

            try (InputStream input = stream) {
                response = factory.newDocumentBuilder()
                        .parse(input);
            }

            NodeList faults =
                    response.getElementsByTagNameNS("*", "Fault");

            if (faults.getLength() > 0) {
                throw new IOException(
                        "SOAP Fault: "
                        + faults.item(0).getTextContent());
            }

            if (statusCode < 200 || statusCode >= 300) {
                throw new IOException(
                        "HTTP status: " + statusCode);
            }

            return response;

        } finally {
            conn.disconnect();
        }
    }

    // =============================================
    // XML HELPER
    // =============================================
    static String getChildText(
            Element parent, String name) {

        NodeList children = parent.getChildNodes();

        for (int i = 0; i < children.getLength(); i++) {
            Node node = children.item(i);

            if (node instanceof Element) {
                Element element = (Element) node;

                String localName = element.getLocalName();

                if (localName == null) {
                    localName = element.getNodeName();

                    int colon = localName.indexOf(':');
                    if (colon >= 0) {
                        localName = localName.substring(colon + 1);
                    }
                }

                if (name.equals(localName)) {
                    return element.getTextContent().trim();
                }
            }
        }

        return null;
    }

    // =============================================
    // INPUT: requestListCustomer
    // =============================================
    static List<Customer> requestListCustomer(
            String studentCode,
            String qCode) throws Exception {

        String params =
                "<studentCode>" + escapeXml(studentCode)
                + "</studentCode>"
                + "<qCode>" + escapeXml(qCode)
                + "</qCode>";

        Document response =
                callSoap("requestListCustomer", params);

        List<Customer> customers = new ArrayList<>();

        NodeList elements =
                response.getElementsByTagName("*");

        for (int i = 0; i < elements.getLength(); i++) {
            Node node = elements.item(i);

            if (!(node instanceof Element)) {
                continue;
            }

            Element element = (Element) node;

            String customerId =
                    getChildText(element, "customerId");

            String purchaseCount =
                    getChildText(element, "purchaseCount");

            String totalSpent =
                    getChildText(element, "totalSpent");

            if (customerId != null
                    && purchaseCount != null
                    && totalSpent != null) {

                Customer customer = new Customer(
                        customerId,
                        Integer.parseInt(purchaseCount),
                        new BigDecimal(totalSpent)
                );

                customers.add(customer);
            }
        }

        return customers;
    }

    // =============================================
    // OUTPUT: submitListCustomer
    // =============================================
    static String submitListCustomer(
            String studentCode,
            String qCode,
            List<Customer> customers) throws Exception {

        StringBuilder params = new StringBuilder();

        params.append("<studentCode>")
                .append(escapeXml(studentCode))
                .append("</studentCode>");

        params.append("<qCode>")
                .append(escapeXml(qCode))
                .append("</qCode>");

        for (Customer c : customers) {
            params.append("<data>")
                    .append("<customerId>")
                    .append(escapeXml(c.customerId))
                    .append("</customerId>")
                    .append("<purchaseCount>")
                    .append(c.purchaseCount)
                    .append("</purchaseCount>")
                    .append("<totalSpent>")
                    .append(c.totalSpent.toPlainString())
                    .append("</totalSpent>")
                    .append("</data>");
        }

        Document response = callSoap(
                "submitListCustomer",
                params.toString());

        NodeList elements =
                response.getElementsByTagNameNS("*", "return");

        if (elements.getLength() == 0) {
            return "No status returned";
        }

        return elements.item(0).getTextContent().trim();
    }

    // =============================================
    // XML ESCAPE
    // =============================================
    static String escapeXml(String value) {
        if (value == null) {
            return "";
        }

        return value.replace("&", "&amp;")
                .replace("<", "&lt;")
                .replace(">", "&gt;")
                .replace("\"", "&quot;")
                .replace("'", "&apos;");
    }
